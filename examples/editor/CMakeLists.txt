cmake_minimum_required(VERSION 3.20)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    if(CMAKE_HOST_APPLE AND NOT CMAKE_SYSTEM_NAME AND NOT CMAKE_TOOLCHAIN_FILE
            AND NOT DEFINED CMAKE_OSX_DEPLOYMENT_TARGET AND "$ENV{MACOSX_DEPLOYMENT_TARGET}" STREQUAL "")
        set(CMAKE_OSX_DEPLOYMENT_TARGET "12.0" CACHE STRING "Minimum macOS deployment target")
    endif()
    project(markdown_editor VERSION 0.2.0 LANGUAGES NONE)
    include("${CMAKE_CURRENT_SOURCE_DIR}/HuxerUIProject.cmake")
    if(HUXERUI_PROJECT_PLAN_ONLY)
        return()
    endif()
    set(HUXERUI_MARKDOWN_BUILD_EXAMPLES OFF)
endif()
huxerui_add_app(markdown_editor SOURCES src/app.cpp platform/desktop/main.cpp
    BUNDLE_NAME "Markdown Writer" BUNDLE_IDENTIFIER "org.huxerui.markdown.editor")
if(TARGET HuxerUI::Markdown)
    huxerui_use_library(markdown_editor TARGET HuxerUI::Markdown)
else()
    huxerui_use_library(markdown_editor TARGET HuxerUI::Markdown PATH "${CMAKE_CURRENT_SOURCE_DIR}/../..")
endif()
if(HUXERUI_LIBRARY_GRAPH_ONLY)
    return()
endif()
target_compile_options(markdown_editor PRIVATE "$<$<CXX_COMPILER_ID:MSVC>:/utf-8>")
set(_editor_default_tests OFF)
if(HUXERUI_MARKDOWN_BUILD_TESTS AND TARGET HuxerUI::testing)
    set(_editor_default_tests ON)
endif()
option(HUXERUI_MARKDOWN_EDITOR_BUILD_TESTS "Build writer interaction tests" ${_editor_default_tests})
if(HUXERUI_MARKDOWN_EDITOR_BUILD_TESTS)
    if(NOT TARGET HuxerUI::testing)
        message(FATAL_ERROR "Writer tests require HUXERUI_BUILD_TESTING_LIBRARY with HuxerUI source")
    endif()
    enable_testing()
    add_executable(markdown_editor_tests tests/editor_test.cpp src/app.cpp)
    add_dependencies(markdown_editor_tests markdown_editor)
    target_link_libraries(markdown_editor_tests PRIVATE HuxerUI::Markdown HuxerUI::testing)
    target_compile_options(markdown_editor_tests PRIVATE "$<$<CXX_COMPILER_ID:MSVC>:/utf-8>")
    if(APPLE)
        set(_editor_resources "$<TARGET_BUNDLE_DIR:markdown_editor>/Contents/Resources/HuxerUI")
    else()
        set(_editor_resources "$<TARGET_FILE_DIR:markdown_editor>/markdown_editor.resources")
    endif()
    add_test(NAME markdown.editor COMMAND markdown_editor_tests "${_editor_resources}")
endif()
