cmake_minimum_required(VERSION 3.20)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    if(CMAKE_HOST_APPLE AND NOT CMAKE_SYSTEM_NAME AND NOT CMAKE_TOOLCHAIN_FILE
            AND NOT DEFINED CMAKE_OSX_DEPLOYMENT_TARGET AND "$ENV{MACOSX_DEPLOYMENT_TARGET}" STREQUAL "")
        set(CMAKE_OSX_DEPLOYMENT_TARGET "12.0" CACHE STRING "Minimum macOS deployment target")
    endif()
    project(markdown_editor VERSION 0.2.2 LANGUAGES NONE)
    include("${CMAKE_CURRENT_SOURCE_DIR}/HuxerUIProject.cmake")
    if(HUXERUI_PROJECT_PLAN_ONLY)
        return()
    endif()
    set(HUXERUI_MARKDOWN_BUILD_EXAMPLES OFF)
endif()
set(_writer_resource_output)
if(ANDROID AND HUXERUI_ANDROID_RESOURCE_OUTPUT_ROOT)
    list(APPEND _writer_resource_output RESOURCE_OUTPUT_DIRECTORY
        "${HUXERUI_ANDROID_RESOURCE_OUTPUT_ROOT}/${ANDROID_ABI}")
endif()
huxerui_add_app(markdown_editor SOURCES src/app.cpp src/main.cpp src/source_editor.cpp
    BUNDLE_NAME "Markdown Writer" BUNDLE_IDENTIFIER "org.huxerui.markdown.editor"
    RESOURCES resources RESOURCE_NAMESPACE markdown_writer
    ${_writer_resource_output})
if(TARGET HuxerUI::Markdown)
    huxerui_use_library(markdown_editor TARGET HuxerUI::Markdown)
else()
    huxerui_use_library(markdown_editor TARGET HuxerUI::Markdown PATH "${CMAKE_CURRENT_SOURCE_DIR}/../..")
endif()
if(HUXERUI_LIBRARY_GRAPH_ONLY)
    return()
endif()
target_compile_options(markdown_editor PRIVATE "$<$<CXX_COMPILER_ID:MSVC>:/utf-8>")
set(_writer_document_default OFF)
if(ANDROID)
    set(_writer_document_default ON)
endif()
option(HUXERUI_MARKDOWN_EDITOR_DOCUMENT_MODE "Enable experimental document editing with the paragraph-layout SDK" ${_writer_document_default})
if(HUXERUI_MARKDOWN_EDITOR_DOCUMENT_MODE)
    add_library(markdown_writer_document STATIC
        ../../src/document_editor_internal.cpp
        ../../experiments/rich_text_editor/editor.cpp
        ../../experiments/rich_text_editor/model.cpp)
    target_link_libraries(markdown_writer_document PUBLIC HuxerUI::Markdown)
    target_include_directories(markdown_writer_document PUBLIC ../../src ../../experiments/rich_text_editor)
    target_compile_definitions(markdown_writer_document PUBLIC HUXERUI_MARKDOWN_EDITOR_DOCUMENT_MODE=1)
    target_compile_options(markdown_writer_document PRIVATE "$<$<CXX_COMPILER_ID:MSVC>:/utf-8>")
    target_link_libraries(markdown_editor PRIVATE markdown_writer_document)
endif()
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR AND CMAKE_SYSTEM_NAME STREQUAL "Linux")
    install(TARGETS markdown_editor RUNTIME DESTINATION bin)
    install(DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/markdown_editor.resources/" DESTINATION bin/markdown_editor.resources)
    install(FILES "${CMAKE_CURRENT_SOURCE_DIR}/platform/linux/markdown-writer.desktop"
        DESTINATION share/applications)
    install(FILES "${CMAKE_CURRENT_SOURCE_DIR}/platform/android/app/src/main/res/mipmap-xxxhdpi/ic_launcher.png"
        DESTINATION share/icons/hicolor/192x192/apps RENAME markdown-writer.png)
    set(CPACK_PACKAGE_NAME "markdown-writer")
    set(CPACK_PACKAGE_VENDOR "HuxerUI")
    set(CPACK_PACKAGE_CONTACT "FarnaHerry <108510510+FarnaHerry@users.noreply.github.com>")
    set(CPACK_PACKAGE_DESCRIPTION_SUMMARY "Markdown Writer editing demo")
    set(CPACK_PACKAGE_VERSION "${PROJECT_VERSION}")
    set(CPACK_GENERATOR "DEB;RPM")
    set(CPACK_DEBIAN_PACKAGE_SHLIBDEPS ON)
    set(CPACK_RPM_PACKAGE_AUTOREQPROV ON)
    include(CPack)
endif()
set(_editor_default_tests OFF)
if(HUXERUI_MARKDOWN_BUILD_TESTS AND TARGET HuxerUI::testing)
    set(_editor_default_tests ON)
endif()
option(HUXERUI_MARKDOWN_EDITOR_BUILD_TESTS "Build writer interaction tests" ${_editor_default_tests})
if(HUXERUI_MARKDOWN_EDITOR_BUILD_TESTS)
    if(NOT TARGET HuxerUI::testing)
        message(FATAL_ERROR "Writer tests require HUXERUI_BUILD_TESTING_LIBRARY with HuxerUI source")
    endif()
    enable_testing()
    add_executable(markdown_editor_tests tests/editor_test.cpp src/app.cpp src/source_editor.cpp)
    add_dependencies(markdown_editor_tests markdown_editor)
    huxerui_add_resources(markdown_editor_tests
        ROOT "${CMAKE_CURRENT_SOURCE_DIR}/resources"
        NAMESPACE "markdown_writer")
    target_link_libraries(markdown_editor_tests PRIVATE HuxerUI::Markdown HuxerUI::testing)
    if(HUXERUI_MARKDOWN_EDITOR_DOCUMENT_MODE)
        target_link_libraries(markdown_editor_tests PRIVATE markdown_writer_document)
    endif()
    target_compile_options(markdown_editor_tests PRIVATE "$<$<CXX_COMPILER_ID:MSVC>:/utf-8>")
    if(APPLE)
        set(_editor_resources "$<TARGET_BUNDLE_DIR:markdown_editor>/Contents/Resources/HuxerUI")
    else()
        set(_editor_resources "$<TARGET_FILE_DIR:markdown_editor>/markdown_editor.resources")
    endif()
    add_test(NAME markdown.editor COMMAND markdown_editor_tests "${_editor_resources}")
    add_executable(markdown_source_editor_tests tests/source_editor_test.cpp src/source_editor.cpp)
    target_link_libraries(markdown_source_editor_tests PRIVATE HuxerUI::testing)
    target_compile_options(markdown_source_editor_tests PRIVATE "$<$<CXX_COMPILER_ID:MSVC>:/utf-8>")
    add_test(NAME markdown.source-editor COMMAND markdown_source_editor_tests)
endif()
